**Message receive handler (per D-03):**
```javascript
ws.onmessage = (e) => {
    try {
        const msg = JSON.parse(e.data);
        if (msg.type === 'message.create') {
            const content = msg.payload?.content || '';
            appendMessage(content, 'server');
        }
    } catch (err) {
        console.error('parse error:', err);
    }
};
```

**appendMessage helper function:**
```javascript
function appendMessage(content, type) {
    const div = document.createElement('div');
    div.className = type; // 'user' or 'server'
    div.textContent = content; // NOT innerHTML (no XSS until Phase 3)
    messages.appendChild(div);
    messages.scrollTop = messages.scrollHeight; // Auto-scroll
}
```

const input = document.getElementById('input');
const sendBtn = document.getElementById('send');
const messages = document.getElementById('messages');

sendBtn.onclick = () => {
    const content = input.value.trim();
    if (!content || ws.readyState !== WebSocket.OPEN) return;
    
    // Send via WebSocket (per D-02)
    ws.send(JSON.stringify({
        type: 'message.send',
        timestamp: Date.now(),
        payload: { content }

    // Display user message (per D-05)
    appendMessage(content, 'user');
